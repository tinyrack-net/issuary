This page documents the complete configuration schema for Tinyrack Auth's `config.yaml` file.

## Full Schema

```yaml
# Application settings
app:
  name: string                      # Application name (default: "Tinyrack Auth")
  host: string                      # Base URL (default: "http://localhost:8080")
  port: number                      # Server port (default: 8080)
  cookie_secret: string             # Session cookie secret (required, min 16 chars)
  
  # JWT settings
  jwt_access_token_ttl: number      # Access token TTL in seconds (default: 3600)
  jwt_refresh_token_ttl: number     # Refresh token TTL in seconds (default: 2592000)
  jwt_key_rotation_enabled: boolean # Enable key rotation (default: true)
  jwt_key_rotation_days: number     # Days between rotations (default: 30)
  jwt_key_overlap_days: number      # Days to keep old keys valid (default: 7)
  
  # Localization
  supported_languages: string[]     # Supported languages (default: ["en"])
  default_language: string          # Default language (default: "auto")
  fallback_language: string         # Fallback language (default: "en")
  
  # Theming
  light_theme: string               # Light theme name (default: "light")
  dark_theme: string                # Dark theme name (default: "dark")
  theme_mode: string                # "light" | "dark" | "system" (default: "system")
  background_url: string            # Background image URL (optional)
  icon_url: string                  # Logo URL (optional)
  
  # Custom text (keyed by language code)
  title: { [lang]: string }         # Login page title (optional)
  subtitle: { [lang]: string }      # Login page subtitle (optional)
  signup_implicit_terms: { [lang]: string }  # Implicit consent notice (optional)
  
  # Signup restrictions
  allowed_signup_emails: string[]   # Email patterns allowed (default: [])
  
  # Proxy
  trust_proxy: boolean | string | string[] | number  # Proxy trust config (default: false)

# Admin server (optional)
admin:
  enabled: boolean                  # Enable admin server (default: false)
  port: number                      # Admin server port (default: 8081)

# Database configuration
database:
  type: "sqlite" | "postgres" | "memory"
  
  # SQLite options
  path: string                      # SQLite database path (default: "test.db")
  
  # PostgreSQL options
  host: string                      # Database host (default: "localhost")
  port: number                      # Database port (default: 5432)
  user: string                      # Database user (default: "test")
  password: string                  # Database password (default: "test")
  name: string                      # Database name (default: "test")

# SMTP configuration (optional)
smtp:
  test: boolean                     # Test mode - log emails (default: false)
  host: string                      # SMTP host (default: "localhost")
  port: number                      # SMTP port (default: 465)
  secure: boolean                   # Use TLS (default: true)
  user: string                      # SMTP username
  password: string                  # SMTP password
  from: string                      # Sender email address

# Basic authentication methods
basic_authentication_methods:
  password:
    enabled: boolean                # Enable password auth (default: true)
    email_verification: boolean     # Require email verification (default: true)
    second_factor:
      required: boolean             # Require 2FA setup (default: false)
    totp:
      enabled: boolean              # Enable TOTP (default: false)
      issuer: string                # TOTP issuer name (optional)
  
  passkey:
    enabled: boolean                # Enable passkey auth (default: false)
    email_verification: boolean     # Require email verification (default: true)
    rp_id: string                   # WebAuthn Relying Party ID (optional)
    origins: string[]               # Allowed origins (optional)

# OAuth authentication methods (social login)
oauth_authentication_methods:
  - type: "google" | "github" | "apple" | "generic_oauth"
    id: string                      # Unique identifier
    enabled: boolean                # Enable this provider (default: false)
    display_name: string            # Display name in UI (optional)
    icon_url: string                # Icon URL (optional)
    client_id: string               # OAuth client ID
    client_secret: string           # OAuth client secret
    scopes: string[]                # OAuth scopes (optional)
    email_conflict_strategy: "auto_link" | "require_link"  # (default: "auto_link")
    
    # Generic OAuth only
    authorization_url: string       # Authorization endpoint
    token_url: string               # Token endpoint
    userinfo_url: string            # UserInfo endpoint
    email_url: string               # Email endpoint (optional)
    response_mode: "query" | "fragment" | "form_post"  # (optional)
    userinfo_mapping:
      id: string                    # Field for user ID
      email: string                 # Field for email
      email_verified: string        # Field for email_verified (optional)
      name: string                  # Field for name (optional)
      picture: string               # Field for picture (optional)

# OAuth client providers (your applications)
providers:
  - id: string                      # Unique identifier
    name: string                    # Display name
    logo_uri: string                # Logo URL (optional)
    client_id: string               # Client ID
    client_secret: string           # Client secret (optional for public clients)
    redirect_uris: string[]         # Allowed redirect URIs
    response_types: string[]        # Allowed response types
    grant_types: string[]           # Allowed grant types
    scope: string                   # Allowed scopes

# Pre-configured users (optional)
users:
  - id: string                      # Unique identifier
    email: string                   # Email address
    password: string                # Password (will be hashed)
    name: string                    # Display name (optional)

# Terms of service (optional)
terms:
  - id: string                      # Unique identifier (lowercase, alphanumeric, -, _)
    required: boolean               # Required agreement (default: true)
    consent_mode: "explicit" | "implicit"  # Consent mode (default: "explicit")
    version: string                 # Version string
    content:
      [lang]:                       # Language code (e.g., "en", "ko")
        title: string               # Display title
        type: "link" | "text"       # Content type
        content: string             # URL or text content

# Account deletion settings
account_deletion:
  enabled: boolean                  # Allow account deletion (default: false)
  retention_period: string          # Retention period (default: "30d")
```

## Environment Variable Interpolation

Use `!env` to reference environment variables:

```yaml
app:
  cookie_secret: !env COOKIE_SECRET

database:
  password: !env DB_PASSWORD
```

## Minimal Configuration

```yaml
app:
  host: http://localhost:8080
  port: 8080
  cookie_secret: !env COOKIE_SECRET

database:
  type: sqlite
  path: ./data/auth.db

basic_authentication_methods:
  password:
    enabled: true
```

## Production Configuration

```yaml
app:
  name: My Auth Server
  host: https://auth.example.com
  port: 8080
  cookie_secret: !env COOKIE_SECRET
  jwt_key_rotation_enabled: true
  supported_languages: [en, ko, ja]
  trust_proxy: true

database:
  type: postgres
  host: !env DB_HOST
  port: 5432
  user: !env DB_USER
  password: !env DB_PASSWORD
  name: !env DB_NAME

smtp:
  host: smtp.example.com
  port: 587
  secure: true
  user: !env SMTP_USER
  password: !env SMTP_PASSWORD
  from: noreply@example.com

basic_authentication_methods:
  password:
    enabled: true
    email_verification: true
    totp:
      enabled: true
      issuer: My App
  passkey:
    enabled: true
    rp_id: example.com

oauth_authentication_methods:
  - type: google
    id: google
    enabled: true
    client_id: !env GOOGLE_CLIENT_ID
    client_secret: !env GOOGLE_CLIENT_SECRET

providers:
  - id: my-web-app
    name: My Web App
    client_id: web-app-client
    client_secret: !env WEB_APP_SECRET
    redirect_uris:
      - https://myapp.example.com/callback
    response_types: [code]
    grant_types: [authorization_code, refresh_token]
    scope: openid profile email

account_deletion:
  enabled: true
  retention_period: 30d

terms:
  - id: privacy-policy
    required: true
    consent_mode: implicit
    version: "1.0.0"
    content:
      en:
        title: Privacy Policy
        type: link
        content: https://example.com/privacy
```

## Type Reference

### AppTheme

Available DaisyUI theme values:

```
light, dark, cupcake, bumblebee, emerald, corporate, synthwave, retro,
cyberpunk, valentine, halloween, garden, forest, aqua, lofi, pastel,
fantasy, wireframe, black, luxury, dracula, cmyk, autumn, business,
acid, lemonade, night, coffee, winter, dim, nord, sunset, caramellatte,
abyss, silk
```

### Duration Format

For `retention_period`:

- `30d` - 30 days
- `24h` - 24 hours
- `60m` - 60 minutes

### trust_proxy Values

| Value | Description |
|-------|-------------|
| `true` | Trust all proxies |
| `false` | Trust no proxies |
| `"loopback"` | Trust loopback addresses |
| `["10.0.0.0/8"]` | Trust specific CIDR ranges |
| `1` | Trust nth hop |

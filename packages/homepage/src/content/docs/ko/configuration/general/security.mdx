```yaml
app:
  # --------------------------------------------------------------------------
  # Security Settings
  # --------------------------------------------------------------------------

  # cookie_secret: A 256-bit (32 bytes) secret key for signing session cookies.
  # IMPORTANT: Generate a unique, random value for production environments.
  # You can generate one using: openssl rand -hex 32
  # WARNING: Changing this value will invalidate all existing sessions.
  cookie_secret: $COOKIE_SECRET

  # jwt_access_token_ttl: Time-to-live for JWT access tokens in seconds.
  # Access tokens are short-lived and used for API authentication.
  # Default: 3600 (1 hour)
  # Recommended: 900-3600 (15 minutes to 1 hour)
  jwt_access_token_ttl: 3600

  # jwt_refresh_token_ttl: Time-to-live for JWT refresh tokens in seconds.
  # Refresh tokens are long-lived and used to obtain new access tokens.
  # Default: 2592000 (30 days)
  # Recommended: 604800-2592000 (7-30 days)
  jwt_refresh_token_ttl: 2592000

  # allowed_signup_emails: List of email patterns allowed to register.
  # Controls who can create new accounts on your authentication server.
  #
  # Pattern syntax:
  #   - '*'                : Allow all emails (public registration)
  #   - 'user@example.com' : Allow only this specific email address
  #   - '*@example.com'    : Allow all emails from the example.com domain
  #   - '*@*.example.com'  : Allow all emails from any subdomain of example.com
  #
  # Multiple patterns can be combined. If any pattern matches, signup is allowed.
  # If this list is empty or not specified, no new signups are allowed.
  allowed_signup_emails:
    # - '*'                    # Allow all emails (public registration)
    - 'admin@example.com'      # Allow specific email
    - '*@example.com'          # Allow all emails from domain

  # --------------------------------------------------------------------------
  # Account Deletion
  # --------------------------------------------------------------------------

  # account_deletion: Whether users can delete their own accounts.
  # When enabled, users can request account deletion from their profile settings.
  # The retention period is configured in the cleanup section.
  # Default: false
  account_deletion: true

```
